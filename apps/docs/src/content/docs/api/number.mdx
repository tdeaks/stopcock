---
title: N (Number)
description: clamp, sum, mean, percentile. Numeric transforms for pipe
---

## Aggregation

```ts
sum(nums: number[]): number
mean(nums: number[]): Option<number>
median(nums: number[]): Option<number>
variance(nums: number[]): Option<number>
standardDeviation(nums: number[]): Option<number>
min(nums: number[]): Option<number>
max(nums: number[]): Option<number>
minMax(nums: number[]): Option<readonly [number, number]>
```

Every partial aggregate also has an explicit `*OrUndefined` variant. Use the
`*NonEmpty` variants when a readonly non-empty tuple proves a result exists,
and the sample-statistic `*AtLeastTwo` variants when two inputs are required.

## Dual

```ts
percentile(nums: number[], p: number): Option<number>
percentileOrUndefined(nums: number[], p: number): number | undefined
percentileNonEmpty(nums: readonly [number, ...number[]], p: number): number
clamp(value: number, min: number, max: number): number
dotProduct(a: number[], b: number[]): number
```

## Predicates

```ts
isEven(n: number): boolean
isOdd(n: number): boolean
```

## Examples

```ts
import { pipe } from '@stopcock/fp'
import * as A from '@stopcock/fp/array'
import * as N from '@stopcock/fp/number'
import * as O from '@stopcock/fp/option'

// stats on response times
const times = [120, 95, 200, 88, 150, 300, 110] as const
pipe(N.mean(times), O.getOrElse(() => 0)) // ~151.9
N.medianNonEmpty(times) // 120
N.standardDeviationNonEmpty(times) // ~68.6
N.percentileNonEmpty(times, 95) // 300

// clamp user input
pipe(userAge, N.clamp(0, 150))

// average score, ignoring zeros
pipe(
  scores,
  A.filter((s) => s > 0),
  N.mean,
)

// feature vector similarity
N.dotProduct([1, 0, 1], [0, 1, 1]) // 1
```
