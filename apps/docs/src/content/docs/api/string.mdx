## Examples

```ts
import { pipe } from '@stopcock/fp'
import * as S from '@stopcock/fp/string'

// slug from title
pipe('  Hello World!  ', S.trim, S.toLowerCase, S.replaceAll(' ', '-'), S.replaceAll('!', ''))
// 'hello-world'

// parse CSV header
pipe('name,email,role', S.split(','))
// ['name', 'email', 'role']

// check file extension
pipe('report.pdf', S.endsWith('.pdf')) // true

// template
pipe('Hello, NAME!', S.replaceAll('NAME', user.name))

// truncate
const preview = S.slice(description, 0, 100) // direct, data-first
const alsoPreview = pipe(description, S.slice(0, 100)) // curried, data-last
```
